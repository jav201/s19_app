"""Probe (E5 corrective, M-L ruling): where does a focus landing meet a shown window with no region?

Spies on S19TuiApp._land_focus and records, at every call, the shown window, the stamp, on_switch,
layout_waits and the region area of the shown window's container. A call with area 0 is a route
where I-D3's layout wait decides the landing. Not a test of the product: it asserts nothing.
"""
from __future__ import annotations

from pathlib import Path

import pytest

from tests.test_b101_unload_focus import (  # noqa: F401
    RAIL_TARGETS,
    case08,
    case_session,
    focus_report,
    open_split_join,
    run,
    settle,
    wait_screen,
)
from tests.support import b100_projects as projects

LOG: list = []


def spy_on(app, label: str) -> None:
    original = app._land_focus

    def spy(trigger, switch, *, on_switch, layout_waits=3):
        key = app._active_screen_key
        cid = "screen_split_join" if key == "split_join" else app.SCREEN_CONTAINER_IDS.get(key)
        try:
            area = app.query_one(f"#{cid}").region.area
        except Exception as exc:  # noqa: BLE001
            area = f"err {type(exc).__name__}"
        try:
            current = type(app.screen).__name__
        except Exception:  # noqa: BLE001
            current = "?"
        LOG.append((label, key, switch, on_switch, layout_waits, area, current))
        print(f"PROBE call {label}: shown={key} stamp={switch} on_switch={on_switch} waits={layout_waits} area={area} screen={current}", flush=True)
        original(trigger, switch, on_switch=on_switch, layout_waits=layout_waits)

    app._land_focus = spy


async def landed(pilot, app, label: str) -> None:
    await settle(pilot, 12)
    print(f"PROBE end  {label}: {focus_report(app)}", flush=True)


def test_probe_fresh_keys(tmp_path: Path) -> None:
    async def scenario(app, pilot, log):
        spy_on(app, "fresh")
        for key, window, _target in RAIL_TARGETS[1:] + RAIL_TARGETS[:1]:
            await pilot.press(key)
            await landed(pilot, app, f"fresh {key}->{window}")

    run(tmp_path, scenario)


def test_probe_fresh_first_action_before_any_pause(tmp_path: Path) -> None:
    async def _drive():
        from s19_app.tui.app import S19TuiApp

        app = S19TuiApp(base_dir=tmp_path)
        async with app.run_test(size=(120, 56)) as pilot:
            spy_on(app, "first")
            app.action_show_screen("mac")
            await landed(pilot, app, "first mac")

    import asyncio

    asyncio.run(_drive())

def test_probe_switch_under_a_modal(tmp_path: Path, case08: Path) -> None:
    async def scenario(app, pilot, log):
        await case_session(pilot, app, case08, mac=True, a2l=True)
        spy_on(app, "modal")
        await pilot.press("question_mark")
        await settle(pilot, 6)
        print(f"PROBE modal open: screen={type(app.screen).__name__}", flush=True)
        app.action_show_screen("mac")
        await landed(pilot, app, "modal mac (under the modal)")
        await pilot.press("escape")
        await landed(pilot, app, "modal closed")

    run(tmp_path, scenario)
