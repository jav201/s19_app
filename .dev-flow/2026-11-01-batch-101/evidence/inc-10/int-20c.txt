=== probe script:
"""Probe (Inc 10a E4 round 2, int-20b): the regions the MAC switch landing of
R-SHELL-FOCUS.2 case /7 sees (``2`` then ``3`` in one burst). A spy wraps
``S19TuiApp._land_focus`` at run time (no product file is changed) and records,
per call, the stamped window, the shown window, ``layout_waits`` and the areas
of ``#screen_mac``, ``#mac_records_list`` and ``#mac_page_prev_button``."""
import os
import pytest

import tests.test_b101_unload_focus as F
from tests.test_b101_unload_focus import case08  # noqa: F401  (module fixture)
from s19_app.tui.app import S19TuiApp


@pytest.mark.parametrize("i", range(6))
def test_probe_burst(tmp_path, case08, i, monkeypatch):  # noqa: F811
    seen = []
    orig = S19TuiApp._land_focus

    def spy(self, trigger, switch, *, on_switch, layout_waits=3):
        areas = tuple(
            self.query_one(f"#{wid}").region.area
            for wid in ("screen_mac", "mac_records_list", "mac_page_prev_button")
        )
        seen.append((switch, self._active_screen_key, layout_waits) + areas)
        return orig(self, trigger, switch, on_switch=on_switch, layout_waits=layout_waits)

    monkeypatch.setattr(S19TuiApp, "_land_focus", spy)

    async def scenario(app, pilot, log):
        await F.case_session(pilot, app, case08, mac=True, a2l=True)
        seen.clear()
        app.action_show_screen("a2l")
        app.action_show_screen("mac")
        await F.wait_screen(pilot, app, "mac")
        await F.settle(pilot, 20)
        line = (
            f"[probe {i}] landing calls (stamp, shown, layout_waits, area screen_mac, "
            f"mac_records_list, mac_page_prev_button)={seen} | final focus={F.focus_report(app)}"
        )
        print(line)
        with open(os.environ["PROBE_OUT"], "a", encoding="utf-8") as out:
            out.write(line + chr(10))

    F.run(tmp_path, scenario)
