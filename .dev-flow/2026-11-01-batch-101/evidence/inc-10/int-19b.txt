"""Probe (Inc 10a E4 rulings): the focus a window switch lands on with nothing
loaded, and where it ends after an image load; and PgDn right after ``3``.
Run from a checkout root with HOME / XDG / LOCALAPPDATA pointed at a temp dir."""
import asyncio
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, str(Path.cwd()))
import s19_app  # noqa: E402
from s19_app.tui.app import S19TuiApp  # noqa: E402

print("s19_app.__file__ =", s19_app.__file__)
CASE01 = Path.cwd() / "examples" / "case_01_basic_valid"


def desc(app):
    w = app.default_screen.focused
    if w is None:
        return "None"
    return f"{type(w).__name__}#{w.id} area={w.region.area}"


async def settle(pilot, n=20):
    for _ in range(n):
        await pilot.pause()


async def switch_then_load(window):
    with tempfile.TemporaryDirectory() as tmp:
        app = S19TuiApp(base_dir=Path(tmp))
        async with app.run_test(size=(118, 54)) as pilot:
            await settle(pilot, 4)
            startup = desc(app)
            app.action_open_split_join_export()
            await settle(pilot)
            await pilot.press("1")
            await settle(pilot)
            if window != "workspace":
                app.action_show_screen(window)
                await settle(pilot)
            landed = desc(app)
            app.load_selected_file(CASE01 / "firmware.s19")
            for _ in range(300):
                await pilot.pause()
                if str(app._busy.file_state).startswith("Loaded "):
                    break
            await settle(pilot, 30)
            print(f"[{window}] startup={startup} | switch landing={landed} | after load={desc(app)}")


async def fresh_load():
    with tempfile.TemporaryDirectory() as tmp:
        app = S19TuiApp(base_dir=Path(tmp))
        async with app.run_test(size=(118, 54)) as pilot:
            await settle(pilot, 4)
            startup = desc(app)
            app.load_selected_file(CASE01 / "firmware.s19")
            for _ in range(300):
                await pilot.pause()
                if str(app._busy.file_state).startswith("Loaded "):
                    break
            await settle(pilot, 30)
            print(f"[fresh] startup={startup} | after load={desc(app)}")


async def mac_pagedown():
    with tempfile.TemporaryDirectory() as tmp:
        app = S19TuiApp(base_dir=Path(tmp))
        async with app.run_test(size=(118, 54)) as pilot:
            await settle(pilot, 4)
            app.load_selected_file(CASE01 / "firmware.s19")
            for _ in range(300):
                await pilot.pause()
                if str(app._busy.file_state).startswith("Loaded "):
                    break
            await settle(pilot, 20)
            app.load_selected_file(CASE01 / "firmware.mac")
            for _ in range(300):
                await pilot.pause()
                if app.current_file is not None and app.current_file.mac_path is not None:
                    break
            await settle(pilot, 20)
            app.mac_records_page_size = 1
            app.update_mac_view()
            await settle(pilot, 10)
            await pilot.press("3")
            await settle(pilot, 20)
            before = app._mac_window_start
            focus = desc(app)
            await pilot.press("pagedown")
            await settle(pilot, 10)
            print(f"[mac PgDn] focus after 3={focus} | _mac_window_start {before} -> {app._mac_window_start}")


async def main():
    await fresh_load()
    for window in ("workspace", "issues", "checks", "a2l", "mac"):
        await switch_then_load(window)
    await mac_pagedown()


asyncio.run(main())
