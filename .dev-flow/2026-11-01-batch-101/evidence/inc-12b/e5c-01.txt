#!/usr/bin/env bash
# Local: the corrective diff and the tree it gives on the base, computed in a
# scratch index and a scratch object directory (the repo's objects as alternates).
set -eu
S="$1"; cd <HOME>/Github/s19_app-b101-inc12b
git diff --binary 2f54c7d -- s19_app tests docs README.md > "$S/i12bc.diff"
rm -rf "$S/objs" "$S/idx"; mkdir -p "$S/objs"
export GIT_OBJECT_DIRECTORY="$S/objs" GIT_ALTERNATE_OBJECT_DIRECTORIES="$(git rev-parse --path-format=absolute --git-common-dir)/objects" GIT_INDEX_FILE="$S/idx"

--- helper source: probe_widths.py (removed after the runs)
"""Read-only probe, Inc 12b E5 corrective (R-1 / QA-3: measure first; R-2: the
A2B card geometry). Drives AT-B94-15's and AT-B94-18's setups and prints the
rail's widths, the painted chain width and the A2B card's rows. Asserts nothing."""
import asyncio

from textual.widgets import Button, Input

from s19_app.tui.app import S19TuiApp
import tests.test_tui_flow_builder_b94 as b94


def _measure(app, label):
    rail = app.query_one("#flow_pipeline_scroll")
    pipe = app.query_one("#flow_pipeline")
    panel = app.query_one("#flow_panel")
    lines = pipe.render().plain.split("\n")
    print(
        f"PROBE {label}: rail.region.width={rail.region.width}"
        f" rail.content_region.width={rail.content_region.width}"
        f" rail.scrollable_content_region.width={rail.scrollable_content_region.width}"
        f" pipeline.region.width={pipe.region.width}"
        f" longest painted line={max(len(line) for line in lines)} lines={len(lines)}"
        f" card_w={panel._card_w} spare={panel._spare} layout_key={panel._layout_key}"
    )
    return lines


def test_probe_rail_widths(tmp_path):
    project = b94._project(tmp_path)

    async def chain(size):
        app = S19TuiApp(base_dir=tmp_path)
        async with app.run_test(size=size) as pilot:
            await pilot.pause()
            await b94._show_flow(app, pilot, project)
            _measure(app, f"{size} 0 cards")
            blocks = (("source", "prg.s19"), ("patch", "hotfix.json"), ("write_out", "out.s19"))
            for n, (kind, ref) in enumerate(blocks, 1):
                await b94._add_ref_block(app, pilot, kind, ref)
                await pilot.pause()
                _measure(app, f"{size} {n} cards")

    async def a2b(size):
        app = S19TuiApp(base_dir=tmp_path)
        async with app.run_test(size=size) as pilot:
            await pilot.pause()
            await b94._show_flow(app, pilot, project)
            await b94._add_ref_block(app, pilot, "source", "prg.s19")
            await pilot.pause()
            app.query_one("#flow_palette_a2b", Button).press()
            await pilot.pause()
            app.query_one("#flow_ref_a", Input).value = "#1"
            app.query_one("#flow_ref_b", Input).value = b94._HOSTILE_SYMBOL.replace("\x00", "")
            app.query_one("#flow_add", Button).press()
            await pilot.pause()
            lines = _measure(app, f"{size} LOAD + A2B (AT-B94-18)")
            for i, line in enumerate(lines):
                print(f"PROBE row {i:2d}: {line!r}")

    asyncio.run(chain((120, 30)))
    asyncio.run(chain((80, 24)))
    asyncio.run(a2b((120, 30)))

--- helper source: probe_settle.py (removed after the runs)
"""Read-only probe, Inc 12b E5 corrective (R-1 / QA-3): does the 120x30 chain of
3 cards settle inside the rail's content width with more frames, and what
narrows the rail? Works on the base tree and on the corrective tree (the
module's chip helper is sync on base, async after Inc 12b). Asserts nothing."""
import asyncio
import inspect

from textual.widgets import Button, Input, Select

from s19_app.tui.app import S19TuiApp
import tests.test_tui_flow_builder_b94 as b94


def _line(app, label):
    rail = app.query_one("#flow_pipeline_scroll")
    pipe = app.query_one("#flow_pipeline")
    panel = app.query_one("#flow_panel")
    lines = pipe.render().plain.split("\n")
    chain = [panel, *panel.ancestors_with_self][1:]
    bars = [
        f"{getattr(w, 'id', None) or type(w).__name__}:w={w.region.width},vbar={w.show_vertical_scrollbar}"
        for w in [panel, *list(panel.ancestors)[:3]]
    ]
    print(
        f"PROBE {label}: rail.region.width={rail.region.width}"
        f" rail.scrollable_content_region.width={rail.scrollable_content_region.width}"
        f" longest painted line={max(len(line) for line in lines)}"
        f" card_w={getattr(panel, '_card_w', '?')} layout_key={getattr(panel, '_layout_key', '?')}"
        f" rail.hbar={rail.show_horizontal_scrollbar} | {' '.join(bars)}"
    )


def test_probe_settle(tmp_path):
    project = b94._project(tmp_path)

    async def add(app, pilot, kind, ref):
        if len(inspect.signature(b94._add_ref_block).parameters) == 3:
            b94._add_ref_block(app, kind, ref)  # base: sets the kind select
        else:
            await b94._add_ref_block(app, pilot, kind, ref)

    async def chain(size):
        app = S19TuiApp(base_dir=tmp_path)
        async with app.run_test(size=size) as pilot:
            await pilot.pause()
            await b94._show_flow(app, pilot, project)
            _line(app, f"{size} 0 cards")
            for n, (kind, ref) in enumerate((("source", "prg.s19"), ("patch", "hotfix.json"), ("write_out", "out.s19")), 1):
                await add(app, pilot, kind, ref)
                await pilot.pause()
                _line(app, f"{size} {n} cards, 1 pause")
            for k in (5, 20, 60):
                for _ in range(k):
                    await pilot.pause()
                _line(app, f"{size} 3 cards, +{k} pauses")
            await pilot.resize_terminal(size[0] + 1, size[1])
            for _ in range(10):
                await pilot.pause()
            await pilot.resize_terminal(*size)
            for _ in range(10):
                await pilot.pause()
            _line(app, f"{size} 3 cards, after a resize out and back")

    asyncio.run(chain((120, 30)))
    asyncio.run(chain((118, 54)))
