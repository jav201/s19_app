def extract(path, names, consts=()):
    tree = ast.parse(path.read_text(encoding="utf-8"))
    out, found = [], set()
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.ClassDef)) and node.name in names:
            out.append(ast.get_source_segment(path.read_text(encoding="utf-8"), node))
            found.add(node.name)
        elif isinstance(node, (ast.Assign, ast.AnnAssign)):
            targets = node.targets if isinstance(node, ast.Assign) else [node.target]
            if any(isinstance(t, ast.Name) and t.id in consts for t in targets):
                out.append(ast.get_source_segment(path.read_text(encoding="utf-8"), node))
                found.update(t.id for t in targets if isinstance(t, ast.Name))
    missing = (set(names) | set(consts)) - found
    assert not missing, missing
    return "\n\n".join(out)

kernel32 = ctypes.WinDLL("kernel32", use_last_error=True)
kernel32.GetFinalPathNameByHandleW.argtypes = [wintypes.HANDLE, wintypes.LPWSTR, wintypes.DWORD, wintypes.DWORD]
kernel32.GetFinalPathNameByHandleW.restype = wintypes.DWORD


def final_path_of_fd(fd):
    buf = ctypes.create_unicode_buffer(1024)
    n = kernel32.GetFinalPathNameByHandleW(msvcrt.get_osfhandle(fd), buf, 1024, 0)
    return buf.value if n else f"<err {ctypes.get_last_error()}>"


def fresh(name):
    base = RUN / name
    root = base / ".s19tool" / "workarea"
    project = root / "p"
    home = project / "flows"
    outside = base / "outside"
    project.mkdir(parents=True)
    outside.mkdir()
    return base, root, project, home, outside


def save(home, root, render=None):
    """save_flow_json's three steps, in its order: guard -> mkdir -> claim."""
    resolved_home, _ = guard(home, root)
    home.mkdir(parents=True, exist_ok=True)
    render = render or (lambda stem: json.dumps({"name": stem, "payload": "CONTENT"}, indent=2))
    return create(home, resolved_home, "z", render)


def listing(folder):
    out = {}
    for dirpath, dirnames, filenames in os.walk(folder, followlinks=False):
        for n in dirnames + filenames:
            p = Path(dirpath) / n
            try:
                st = os.lstat(p)
            except OSError:
                continue
            kind = "link" if (stat.S_ISLNK(st.st_mode) or getattr(st, "st_file_attributes", 0) & 0x400) else (
                "dir" if stat.S_ISDIR(st.st_mode) else f"file {st.st_size}B nlink={st.st_nlink}")
            out[str(p.relative_to(folder))] = kind
    return out


def run(label, fn):
    print(f"\n=== {label}")
    try:
        result = fn()
        print("  result:", result)
    except WCE as exc:
        print("  raised WorkareaContainmentError:", str(exc)[:160])
    except Exception as exc:  # noqa: BLE001
        import traceback
        print("  raised", type(exc).__name__, str(exc)[:160])
        print("  at:", traceback.extract_tb(exc.__traceback__)[-1][:3])


real_open, real_lstat = os.open, os.lstat


def patched(open_fn=None, lstat_fn=None):
    class _P:
        def __enter__(self):
            if open_fn:
                os.open = open_fn
            if lstat_fn:
                os.lstat = lstat_fn

        def __exit__(self, *a):
            os.open, os.lstat = real_open, real_lstat
    return _P()

# P1 a dangling junction put in at the FREE name between lexists and the create (real follow?)
def p1():
    base, root, project, home, outside = fresh("p1")
    state = {}

    def op(path, flags, *a, **k):
        c = Path(os.fspath(path))
        if not state and c.parent == home and c.suffix == ".json":
            state["x"] = 1
            (outside / "t.json").mkdir()
            _winapi.CreateJunction(str(outside / "t.json"), str(c))
            (outside / "t.json").rmdir()  # now dangling
        return real_open(path, flags, *a, **k)
    with patched(open_fn=op):
        try:
            return save(home, root)
        finally:
            print("  after:", listing(base))
run("P1 dangling junction put in at the free name (real Windows create)", p1)

# P3 the home swapped out before the create and BACK inside the check (after its lstat)
def p3():
    base, root, project, home, outside = fresh("p3")
    state = {}

    def op(path, flags, *a, **k):
        c = Path(os.fspath(path))
        if "fd" not in state and c.parent == home and c.suffix == ".json":
            home.rename(project / "flows_aside")
            _winapi.CreateJunction(str(outside), str(home))
            fd = real_open(path, flags, *a, **k)
            state["fd"], state["cand"] = fd, c
            return fd
        return real_open(path, flags, *a, **k)

    def ls(path, *a, **k):
        st = real_lstat(path, *a, **k)
        if "fd" in state and "back" not in state and Path(os.fspath(path)) == state["cand"]:
            state["back"] = 1
            print("  fd final path at the check:", final_path_of_fd(state["fd"]), "nlink", st.st_nlink)
            os.rmdir(home)  # remove the junction
            (project / "flows_aside").rename(home)
        return st
    with patched(open_fn=op, lstat_fn=ls):
        try:
            return save(home, root)
        finally:
            print("  after:", listing(base))
            o = outside / "z.json"
            if o.exists():
                print("  outside/z.json bytes:", o.read_bytes()[:60])
run("P3 home swapped out before the create, back inside the check", p3)

# P9 the home swapped to O before the create (refused by the real-path clause), then
# re-pointed at V (a folder holding a victim z.json) after the close, before the cleanup unlink
def p9():
    base, root, project, home, outside = fresh("p9")
    victim = base / "victim"
    victim.mkdir()
    (victim / "z.json").write_bytes(b"VICTIM")
    state = {}

    def op(path, flags, *a, **k):
        c = Path(os.fspath(path))
        if "fd" not in state and c.parent == home and c.suffix == ".json":
            home.rename(project / "flows_aside")
            _winapi.CreateJunction(str(outside), str(home))
            fd = real_open(path, flags, *a, **k)
            state["fd"] = fd
            return fd
        return real_open(path, flags, *a, **k)

    def cl(fd):
        real_close(fd)
        if fd == state.get("fd") and "re" not in state:
            state["re"] = 1
            os.rmdir(home)
            _winapi.CreateJunction(str(victim), str(home))
    os.close = cl
    try:
        with patched(open_fn=op):
            return save(home, root)
    finally:
        os.close = real_close
        print("  after:", listing(base))
run("P9 refused claim, home re-pointed at V before the cleanup unlink", p9)
